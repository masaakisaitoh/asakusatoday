# サイト全体UI多言語対応 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** ヘッダーの言語セレクタ切替で、記事本文だけでなくUIチューム（ナビ・ボタン・ラベル・エラーメッセージ）も6言語に追従させる。

**Architecture:** `utils/i18n/uiStrings.ts` に `Record<TranslationLocale, Record<UiStringKey, string>>` 型の辞書を定義し、`composables/useUiText.ts` が既存の `useArticleLocale()` のlocale stateを読んで `t(key, params?)` を返す。各ページ/コンポーネントはハードコードされた文言を `t()` 呼び出しに置き換える。

**Tech Stack:** Vue 3 / Nuxt 3（auto-import前提。composableは他ファイルから未importのまま呼び出し、Nuxtがビルド時に解決する）、Vitest + @vue/test-utils（コンポーネント単体テスト）、@nuxt/test-utils/e2e（E2Eテスト）。

## Global Constraints

- 辞書のキー網羅性はTypeScriptの型（`Record<TranslationLocale, Record<UiStringKey, string>>`）で保証する。ランタイムの網羅性チェックは書かない。
- `composables/`配下の新規composableは、既存の`useArticleLocale`と同じパターン（他のcomposableをbareで呼び出し、importしない）に従う。Nuxtのauto-importが本番ビルドで解決する。
- テストで`useState` / `useArticleLocale` / `useUiText`をグローバル参照する箇所は、既存の`layouts/default.test.ts`の`vi.stubGlobal(...)`パターンをそのまま踏襲する。
- 管理画面（`pages/admin/*`）、`utils/countries.ts`の国名データ、サーバー側でthrowされるエラーメッセージ（`session.ts`等）はスコープ外。変更しない。
- 英語(`en`)の文言は、既存のハードコード文言と完全一致させる（変わるのは日本語だったcreate/import/loginのボタン文言のみ）。これにより`WeatherCard.test.ts`・`tests/e2e/map.test.ts`など、英語文言をアサートしている既存テストへの影響を最小化する。

import type { UiStringKey } from '../utils/i18n/uiStrings'
import { UI_STRINGS } from '../utils/i18n/uiStrings'

export function useUiText() {
  const { locale } = useArticleLocale()

  function t(key: UiStringKey, params?: Record<string, string | number>): string {
    let text = UI_STRINGS[locale.value][key]
    if (params) {
      for (const [name, value] of Object.entries(params)) {
        text = text.replaceAll(`{${name}}`, String(value))
      }
    }
    return text
  }

  return { t }
}

  await navigateTo('/login')
}

const regenerating = ref(false)

async function regenerateAvatar(): Promise<void> {
  regenerating.value = true
  await $fetch('/api/user/avatar/regenerate', { method: 'POST' })
  await refresh()
  regenerating.value = false
}

async function logout(): Promise<void> {
  await $fetch('/api/auth/logout', { method: 'POST' })
  await navigateTo('/login')
}

function formatJoinedDate(createdAt: string): string {
  return createdAt.slice(0, 10)
}

function formatAddress(address: string): string {
  return `${address.slice(0, 6)}...${address.slice(-4)}`
}

const genderLabelKeys: Record<string, 'profile.genderMale' | 'profile.genderFemale' | 'profile.genderOther' | 'profile.notSet'> = {
  male: 'profile.genderMale',
  female: 'profile.genderFemale',
  other: 'profile.genderOther',
  unspecified: 'profile.notSet'
}

function genderLabel(gender: string | null): string {
  if (!gender) return t('profile.notSet')
  return t(genderLabelKeys[gender] ?? 'profile.notSet')
}

function nationalityLabel(code: string | null): string {
  if (!code) return t('profile.notSet')
  return COUNTRIES.find((c) => c.code === code)?.name ?? code
}

type Mode = 'view' | 'edit'
const mode = ref<Mode>('view')

const profileSchema = computed(() =>
  z.object({
    userName: z.string().regex(/^[A-Za-z0-9_-]{3,32}$/, t('profile.usernameHint')),
    gender: z.enum(['male', 'female', 'other', 'unspecified']),
    birthYear: z.number().int().min(1900).max(new Date().getFullYear()).nullable(),
    nationality: z.string().length(2).nullable()
  })
)
type ProfileFormState = z.infer<ReturnType<typeof profileSchema.value.parse> extends never ? typeof profileSchema.value : typeof profileSchema.value>

const formState = ref<{
  userName: string
  gender: 'male' | 'female' | 'other' | 'unspecified'
  birthYear: number | null
  nationality: string | null
}>({
  userName: '',
  gender: 'unspecified',
  birthYear: null,
  nationality: null
})

const genderOptions = computed(() => [
  { label: t('profile.genderMale'), value: 'male' },
  { label: t('profile.genderFemale'), value: 'female' },
  { label: t('profile.genderOther'), value: 'other' },
  { label: t('profile.genderPreferNotToSay'), value: 'unspecified' }
])

const saving = ref(false)
const submitError = ref('')

function startEdit(): void {
  if (!user.value) return
  formState.value = {
    userName: user.value.user_name,
    gender: (user.value.gender as typeof formState.value.gender) ?? 'unspecified',
    birthYear: user.value.birth_year,
    nationality: user.value.nationality
  }
  submitError.value = ''
  mode.value = 'edit'
}

function cancelEdit(): void {
  mode.value = 'view'
}

async function onSubmit(event: FormSubmitEvent<typeof formState.value>): Promise<void> {
  saving.value = true
  submitError.value = ''
  try {
    await $fetch('/api/user/profile', { method: 'PATCH', body: event.data })
    await refresh()
    mode.value = 'view'
  } catch (e: any) {
    submitError.value = e?.statusCode === 409 ? t('profile.usernameTaken') : t('profile.genericError')
  } finally {
    saving.value = false
  }
}
</script>

        </div>
      </div>

      <UForm v-else :schema="profileSchema" :state="formState" class="flex flex-col gap-4" @submit="onSubmit">
        <UFormField :label="t('profile.username')" name="userName">
          <UInput v-model="formState.userName" />
        </UFormField>
        <UFormField :label="t('profile.gender')" name="gender">

            :items="COUNTRIES"

            label-key="name"
            :placeholder="t('profile.nationalityPlaceholder')"

            {{ t('profile.cancel') }}
